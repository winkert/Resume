%create data structures outside of the document and then use directlua later on
\begin{luacode*}
    local tex_special_characters = {
        ["\\"] = "\\textbackslash{}",
        ["{"] = "\\{",
        ["}"] = "\\}",
        ["$"] = "\\$",
        ["&"] = "\\&",
        ["#"] = "\\#",
        ["%"] = "\\%",
        ["_"] = "\\_",
        ["^"] = "\\textasciicircum{}",
        ["~"] = "\\textasciitilde{}"
    }

    function tex_escape(value)
        return (tostring(value):gsub("([\\{}$&#%%_^~])", function(character)
            return tex_special_characters[character]
        end))
    end

    function tex_escape_with_linebreaks(value)
        local escaped = tex_escape(value):gsub("\r\n", "\n"):gsub("\r", "\n")
        return (escaped:gsub("\n+", "\\\\"))
    end

    resume_data = require("data\\resume_data")
\end{luacode*}

\begin{document}

\section*{Summary}
\normalsize
\directlua{tex.print(tex_escape_with_linebreaks(resume_data.summary))}\\
\hrul
\section*{Career History}
\directlua{
    for i, company in ipairs(resume_data.career_history) do
        tex.print("\\company{" .. tex_escape(company.company) .. "}{" .. tex_escape(company.period) .. "}\\\\")
        for j, position in ipairs(company.positions) do
            tex.print("\\position{" .. tex_escape(position.title) .. "}{" .. tex_escape(position.period) .. "}\\\\")
            for k, accomplishment in ipairs(position.accomplishments) do
                if accomplishment.header ~= "" then
                    tex.print("\\subsection*{" .. tex_escape(accomplishment.header) .. "}")
                end
                tex.print("\\begin{itemize}")
                for l, item in ipairs(accomplishment.items) do
                    tex.print("\\item " .. tex_escape(item))
                end
                tex.print("\\end{itemize}")
            end
        end
    end
} 
\hrul
% in my dataset, "skills" are broken out by sections
\directlua{
    for i, skill_section in ipairs(resume_data.skills) do
        tex.print("\\section*{" .. tex_escape(skill_section.name) .. "}")
        tex.print("\\begin{itemize}")
        for j, item in ipairs(skill_section.items) do
            tex.print("\\item " .. tex_escape(item))
        end
        tex.print("\\end{itemize}")
    end
}
